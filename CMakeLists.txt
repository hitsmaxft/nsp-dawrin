cmake_minimum_required(VERSION 3.20)
project(nsp_dawrin_services VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Portable service library only. The native macOS runner remains in Makefile.
add_library(nsp_services STATIC src/compat/ipc.c src/compat/clock.c src/compat/time_service.c src/compat/settings_service.c)
target_include_directories(nsp_services PUBLIC src/compat src/backend)
if(WIN32)
    target_sources(nsp_services PRIVATE src/backend/windows/backend_windows.c)
    target_link_libraries(nsp_services PUBLIC bcrypt)
elseif(APPLE)
    target_sources(nsp_services PRIVATE src/backend/posix/common.c src/backend/macos/backend_macos.c)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(nsp_services PRIVATE src/backend/posix/common.c src/backend/linux/backend_linux.c)
else()
    message(FATAL_ERROR "No host service backend for ${CMAKE_SYSTEM_NAME}")
endif()
if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(nsp_services PUBLIC Threads::Threads)
endif()
if(MSVC)
    target_compile_options(nsp_services PRIVATE /W4)
else()
    target_compile_options(nsp_services PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(name backend ipc ipc_objects time_service)
        add_executable(test_${name} tests/${name}.c)
        target_link_libraries(test_${name} PRIVATE nsp_services)
        if(MSVC)
            target_compile_options(test_${name} PRIVATE /UNDEBUG)
        else()
            target_compile_options(test_${name} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${name} COMMAND test_${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 15)
    endforeach()
    set_tests_properties(time_service PROPERTIES ENVIRONMENT "TZ=UTC")
    add_test(NAME time_service_fixed COMMAND test_time_service)
    set_tests_properties(time_service_fixed PROPERTIES TIMEOUT 15 ENVIRONMENT "TZ=UTC;DM_FIXED_TIME=1704067200")
endif()
